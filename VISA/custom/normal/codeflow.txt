    - train_dreambooth.py: training procedure for UNET and text encoder
    
          accelerate launch Diffusers/train_dreambooth.py \
          --pretrained_model_name_or_path="CompVis/stable-diffusion-v1-4" \
          --instance_data_dir="VISA/candle" \
          --output_dir="results/VISA/candle" \
          --instance_prompt="a photo of sks anomalous candle" \
          --resolution=512 \
          --train_batch_size=1 \
          --gradient_accumulation_steps=1 \
          --learning_rate=5e-6 \
          --lr_scheduler="constant" \
          --lr_warmup_steps=0 \
          --max_train_steps=400 \
          --push_to_hub
